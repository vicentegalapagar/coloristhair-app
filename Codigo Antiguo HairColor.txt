// Métodos para estilistas getStylists() { return JSON.parse(localStorage.getItem('newhair_stylists')); }

addStylist(stylist) { const stylists = this.getStylists(); stylist.id = this._generateId(stylists); stylists.push(stylist); localStorage.setItem('newhair_stylists', JSON.stringify(stylists)); return stylist; }

findStylistByEmail(email) { const stylists = this.getStylists(); return stylists.find(s => s.email === email); }

// Métodos para clientes getClients() { return JSON.parse(localStorage.getItem('newhair_clients')); }

addClient(client) { const clients = this.getClients(); client.id = this._generateId(clients); client.createdAt = new Date().toISOString(); client.history = []; clients.push(client); localStorage.setItem('newhair_clients', JSON.stringify(clients)); return client; }

updateClient(client) { const clients = this.getClients(); const index = clients.findIndex(c => c.id === client.id); if (index !== -1) { clients[index] = client; localStorage.setItem('newhair_clients', JSON.stringify(clients)); return true; } return false; }

findClientById(id) { const clients = this.getClients(); return clients.find(c => c.id === id); }

searchClients(query) { const clients = this.getClients(); const lowerQuery = query.toLowerCase(); return clients.filter(c => c.name.toLowerCase().includes(lowerQuery) || c.phone.includes(query) || c.email.toLowerCase().includes(lowerQuery) ); }

addClientService(clientId, service) { const clients = this.getClients(); const client = clients.find(c => c.id === clientId); if (client) { service.id = this._generateId(client.history); service.date = new Date().toISOString(); client.history.push(service); localStorage.setItem('newhair_clients', JSON.stringify(clients)); return service; } return null; }

// Métodos para productos getProducts() { return JSON.parse(localStorage.getItem('newhair_products')); }

addProduct(product) { const products = this.getProducts(); product.id = this._generateId(products); products.push(product); localStorage.setItem('newhair_products', JSON.stringify(products)); return product; }

// Métodos auxiliares _generateId(items) { return items.length > 0 ? Math.max(...items.map(i => i.id)) + 1 : 1; }

}

init() { // Verificar si hay un usuario logueado en sessionStorage const user = sessionStorage.getItem('newhair_current_user'); if (user) { this.currentUser = JSON.parse(user); } }

login(email, password) { const stylist = db.findStylistByEmail(email); if (stylist && stylist.password === password) { this.currentUser = stylist; sessionStorage.setItem('newhair_current_user', JSON.stringify(stylist)); return true; } return false; }

logout() { this.currentUser = null; sessionStorage.removeItem('newhair_current_user'); }

register(stylist) { // Verificar si el email ya existe if (db.findStylistByEmail(stylist.email)) { return { success: false, message: 'El email ya está registrado' }; }

// Añadir el nuevo estilista const newStylist = db.addStylist(stylist); return { success: true, stylist: newStylist };

}

isAuthenticated() { return this.currentUser !== null; }

isAdmin() { return this.isAuthenticated() && this.currentUser.role === 'admin'; }

}
